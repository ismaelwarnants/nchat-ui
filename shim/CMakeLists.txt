# Builds upstream nchat (unmodified, via add_subdirectory) plus `nchatcore`: a static
# library containing upstream main.cpp + theme.cpp and our replacement Ui class.
# Driven by ../build.rs, but can also be built standalone with cmake.

cmake_minimum_required(VERSION 3.16 FATAL_ERROR)
project(nchatui_shim LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

set(NCHAT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../nchat")

# Upstream build options we don't need
set(HAS_TESTS OFF CACHE BOOL "Build tests" FORCE)

add_subdirectory(${NCHAT_DIR} nchat)

# Variables defined inside nchat's scope. Feature flags too: nchat may override the cache
# values internally (e.g. HAS_DYNAMICLOAD is forced OFF for static builds).
foreach(VAR HAS_MULTIPROTOCOL HAS_DYNAMICLOAD HAS_EMOJIWIDTH HAS_COREDUMP HAS_SHARED_LIBS
            HAS_DUMMY HAS_TELEGRAM HAS_WHATSAPP HAS_SIGNAL)
  get_directory_property(${VAR} DIRECTORY ${NCHAT_DIR} DEFINITION ${VAR})
endforeach()
message(STATUS "nchat-ui: shared=${HAS_SHARED_LIBS} dynamicload=${HAS_DYNAMICLOAD} "
               "whatsapp=${HAS_WHATSAPP} telegram=${HAS_TELEGRAM} signal=${HAS_SIGNAL}")
get_directory_property(GO_VERSION DIRECTORY ${NCHAT_DIR} DEFINITION GO_VERSION)
get_directory_property(GO_VERSION_MIN DIRECTORY ${NCHAT_DIR} DEFINITION GO_VERSION_MIN)
get_directory_property(NCHAT_LIBDIR DIRECTORY ${NCHAT_DIR} DEFINITION CMAKE_INSTALL_LIBDIR)
if(NOT NCHAT_LIBDIR)
  set(NCHAT_LIBDIR "lib")
endif()

# Copy upstream main.cpp (byte-identical) out of nchat/src, so that its `#include "ui.h"`
# resolves to our ui.h rather than to nchat/src/ui.h sitting next to it.
set(UPSTREAM_MAIN "${CMAKE_CURRENT_BINARY_DIR}/upstream/main.cpp")
configure_file("${NCHAT_DIR}/src/main.cpp" "${UPSTREAM_MAIN}" COPYONLY)

add_library(nchatcore STATIC
  ${UPSTREAM_MAIN}
  ${NCHAT_DIR}/src/theme.cpp
  ui.cpp
  ui.h
  core.h
)
set_target_properties(nchatcore PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/out")
set_source_files_properties(${UPSTREAM_MAIN} PROPERTIES COMPILE_DEFINITIONS "main=nchat_main")

# Order matters: our dir first so "ui.h" is ours.
target_include_directories(nchatcore PRIVATE
  "${CMAKE_CURRENT_SOURCE_DIR}"
  "${NCHAT_DIR}/lib/common/src"
  "${NCHAT_DIR}/lib/ncutil/src"
  "${NCHAT_DIR}/src"
  "${CMAKE_CURRENT_BINARY_DIR}/nchat/generated"
)

# Mirror the compile definitions upstream gives the nchat executable
target_compile_definitions(nchatcore PRIVATE
  CMAKE_INSTALL_LIBDIR="${NCHAT_LIBDIR}"
  GO_VERSION="${GO_VERSION}"
  GO_VERSION_MIN="${GO_VERSION_MIN}"
)
foreach(FEATURE HAS_MULTIPROTOCOL HAS_DYNAMICLOAD HAS_EMOJIWIDTH HAS_COREDUMP)
  if(${FEATURE})
    target_compile_definitions(nchatcore PRIVATE ${FEATURE}="${${FEATURE}}")
  endif()
endforeach()
foreach(PROTO_PAIR "HAS_DUMMY;duchat" "HAS_TELEGRAM;tgchat" "HAS_WHATSAPP;wmchat" "HAS_SIGNAL;sgchat")
  list(GET PROTO_PAIR 0 FEATURE)
  list(GET PROTO_PAIR 1 PROTO_LIB)
  if(${FEATURE})
    target_compile_definitions(nchatcore PRIVATE ${FEATURE}="${${FEATURE}}")
    target_include_directories(nchatcore PRIVATE "${NCHAT_DIR}/lib/${PROTO_LIB}/src")
    if(NOT HAS_DYNAMICLOAD)
      target_link_libraries(nchatcore PUBLIC ${PROTO_LIB})
    endif()
  endif()
endforeach()
if(CMAKE_BUILD_TYPE STREQUAL "Release")
  target_compile_definitions(nchatcore PRIVATE NCHAT_BUILD_RELEASE=1)
endif()

target_link_libraries(nchatcore PUBLIC ncutil)

# themes.h is generated by a custom command in nchat's directory; building the stock
# nchat target produces it (and gives us the terminal `nchat` for account setup).
add_dependencies(nchatcore nchat)

# Backends are dlopen()ed at runtime, not linked, so build them explicitly.
foreach(PROTO_LIB duchat tgchat wmchat sgchat)
  if(TARGET ${PROTO_LIB})
    add_dependencies(nchatcore ${PROTO_LIB})
  endif()
endforeach()

# Link probe: a throwaway executable linking nchatcore. Building it proves the whole
# static dependency closure links, and its link line in build.ninja tells build.rs the
# exact archives + system libraries (in order) to give rustc.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/linkprobe.cpp"
  "int nchat_main(int argc, char* argv[]);\nint main(int argc, char* argv[]) { return nchat_main(argc, argv); }\n")
add_executable(nchatui_linkprobe "${CMAKE_CURRENT_BINARY_DIR}/linkprobe.cpp")
target_link_libraries(nchatui_linkprobe PRIVATE nchatcore)
set_target_properties(nchatui_linkprobe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/out")

# Tell build.rs where things ended up
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/out/paths.txt"
  "NCHAT_BIN_DIR=${CMAKE_CURRENT_BINARY_DIR}/nchat/bin\n"
  "NCHAT_LIB_DIR=${CMAKE_CURRENT_BINARY_DIR}/nchat/${NCHAT_LIBDIR}\n"
  "SHARED_LIBS=${HAS_SHARED_LIBS}\n"
  "WHATSAPP=${HAS_WHATSAPP}\nTELEGRAM=${HAS_TELEGRAM}\nSIGNAL=${HAS_SIGNAL}\n"
)
